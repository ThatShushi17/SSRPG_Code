func Reduplicate(str, int)
    ?int = 0
        return ""
    ?int = 1
        return str
    
    var temp = ""
    for i = 0..int-1
        temp += str
    
    return temp

func RemoveFromArr(arr, toBeDel)
    var tBDC = toBeDel.Count() - 1

    ?tBDC = 0
        arr.RemoveAt(toBeDel[0])
    :?tBDC > 0
        for i = tBDC .. 0
            arr.RemoveAt(toBeDel[i])

func SetCanvasSprite(canvas, asciiart)
    var lines = Split(asciiart, "\n")

    for y = 0 .. lines.Count() - 1
        var line = lines[y]
        for x = 0 .. Size(line) - 1
            var char = Sub(line, x, 1)
                ?char ! "#"
                    canvas.Set(x,y,char)

var lerpArray = []
func LerpComps ()
    ?lerpArray.Count() = 0
        return

    var toBeDel = []
    for compData : lerpArray
        var elem = compData[0]
        var lerp_delta = compData[1]
        var endX = compData[2][0]
        var endY = compData[2][1]
        var currX = compData[3][0]
        var currY = compData[3][1]
        var startT = compData[4]

        ?totaltime >= startT
            var newX = Lerp(currX, endX, lerp_delta)
            var newY = Lerp(currY, endY, lerp_delta)
            
            ?newX - currX > 0
                elem.x = Ceil(newX)
            :?newX - currX < 0
                elem.x = Floor(newX)
            
            ?newY - currY > 0
                elem.y = Ceil(newY)
            :?newY - currY < 0
                elem.y = Floor(newY)
            
            compData[3][0] = newX
            compData[3][1] = newY
            
            ?elem.x = endX & elem.y = endY
                toBeDel.Add(lerpArray.IndexOf(compData))
    
    RemoveFromArr(lerpArray, toBeDel)

var fadeArray = []
func FadeCanvii()
    ?fadeArray.Count() = 0
        return

    var toBeDel = []
    for compData : fadeArray
        var canvas = compData[0]
        var greys = compData[1]
        var startT = compData[2]
        var delT = totaltime - startT

        ?delT >= 0
            canvas.SetFG(greys[delT])

            ?delT >= greys.Count() - 1
                toBeDel.Add(fadeArray.IndexOf(compData))

    RemoveFromArr(fadeArray, toBeDel)

func RenderDisplay(disp, curr, goal)
    var goalLen = Size(goal + "")
    var currLen = Size(curr + "")
    // var delLen = goalLen - currLen
    var message = " / " + (goal + "")

    for i = 0 .. currLen - 1
        var char = Sub(curr + "", i, 1)
        disp.Set(i, 0, char)

    for i = 0 .. Size(message) - 1
        var char = Sub(message + "", i, 1)
        disp.Set(i + goalLen, 0, char)

func RenderProgBar(bar, curr, goal)
    var prog = curr / (goal * 1.0)
    var progUnits = prog * bar.w
    var intPart = Floor(progUnits)
    var fractPart = Floor((progUnits-intPart)*10)

    ?intPart ! 0
        for i = 1 .. Min(intPart, bar.w)
            bar.SetFG(i - 1, 0, #fff)
    ?intPart < bar.w
        bar.SetFG(intPart, 0, linear_greys_10[fractPart])

func RenderProgress(disp, bar, curr, goal)
    disp.Set("#")
    for i = 0 .. bar.w - 1
        bar.SetFG(i, 0, #000)

    ?currQuestProg >= 0
        RenderDisplay(disp, curr, goal)
        RenderProgBar(bar, curr, goal)
